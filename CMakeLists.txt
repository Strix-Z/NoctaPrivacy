cmake_minimum_required(VERSION 3.21)
project(NoctaPrivacy VERSION 0.1.1 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "NoctaPrivacy requires Windows x64 and Visual Studio MSVC.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build x64, not Win32/x86.")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set(SPOUT2_DIR "${CMAKE_SOURCE_DIR}/external/Spout2" CACHE PATH "Spout2 source checkout")
if(NOT EXISTS "${SPOUT2_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Spout2 is missing. Run build.ps1 (it downloads Spout2) or set SPOUT2_DIR.")
endif()
set(SPOUT_BUILD_SPOUTDX ON CACHE BOOL "" FORCE)
set(SPOUT_BUILD_LIBRARY OFF CACHE BOOL "" FORCE)
set(SPOUT_BUILD_CMT ON CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
add_subdirectory("${SPOUT2_DIR}" "${CMAKE_BINARY_DIR}/spout")

add_executable(NoctaPrivacy src/main.cpp)
target_compile_definitions(NoctaPrivacy PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN SPOUT_BUILD_STATIC)
target_compile_options(NoctaPrivacy PRIVATE /W4 /EHsc /permissive- /utf-8)
target_include_directories(NoctaPrivacy PRIVATE "${SPOUT2_DIR}/SPOUTSDK/SpoutDirectX/SpoutDX")
target_link_libraries(NoctaPrivacy PRIVATE SpoutDX_static d3d11 dxgi d3dcompiler windowsapp user32)
set_target_properties(NoctaPrivacy PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
